local isResultValidData = {}
local gsubData = {}

for k, gsub in pairs(data) do
	local lower = k:lower()

	if lower:sub(1, 7) ~= "chroma " and lower:sub(1, 3) ~= "c. " then
		isResultValidData[lower] = gsub
	else
		gsubData[lower] = gsub
		gsubData[lower:gsub("^c%.%s*", "chroma ")] = gsub
	end
end

local function gsub(gsubCondition)
	if gsubCondition then
		return tostring(gsubCondition):lower():gsub("^%s+", ""):gsub("%s+$", ""):gsub("%s*[x×]%s*%d+$", ""):gsub("%s*%(%d+%)$", ""):gsub("%s*%[%d+%]$", ""):gsub("%s+%d+$", ""):gsub("^%s+", ""):gsub("%s+$", "")
	end

	return ""
end
local function isUpdateSizeYFlagValid(resultFlag)
	return resultFlag:sub(1, 7) == "chroma " or (resultFlag:sub(1, 3) == "c. " or (resultFlag:sub(1, 2) == "c " or (resultFlag:match("%(chroma%)") ~= nil or (resultFlag:match("%[chroma%]") ~= nil or (resultFlag:match("%schroma$") ~= nil or resultFlag:match("%schroma%s") ~= nil)))))
end
local function isUpdateSizeYResultValid(resultCondition, flag)
	local textState = nil
	local secondaryTextState = nil
	local alternateTextState = nil

	if resultCondition then
		local textResult = gsub(resultCondition)

		if not flag and not isUpdateSizeYFlagValid(textResult) then
			return isResultValidData[textResult] or 0
		end

		local text = textResult:gsub("^c%.%s*", "", textState, secondaryTextState, alternateTextState):gsub("^c%s+", "", textState, secondaryTextState, alternateTextState):gsub("^chroma%s+", "", textState, secondaryTextState, alternateTextState):gsub("%s*%(chroma%)%s*", " ", textState, secondaryTextState, alternateTextState):gsub("%s*%[chroma%]%s*", " ", textState, secondaryTextState, alternateTextState):gsub("%s+chroma%s*", " ", textState, secondaryTextState, alternateTextState):gsub("%s+chroma$", "", textState, secondaryTextState, alternateTextState):gsub("%s+", " ", textState, secondaryTextState, alternateTextState):gsub("^%s+", "", textState, secondaryTextState, alternateTextState):gsub("%s+$", "", textState, secondaryTextState, alternateTextState)

		return gsubData["chroma " .. text] or (gsubData[textResult] or (gsubData["c. " .. text] or 0))
	end

	return 0
end

local function format(formatNumber)
	if not (formatNumber >= 1000000) then
		if not (formatNumber >= 1000) then
			return string.format("%.2f", formatNumber)
		end

		return string.format("%.2fK", formatNumber / 1000)
	end

	return string.format("%.2fM", formatNumber / 1000000)
end

local connection = nil

local function secondaryUpdateHeartbeatConnection(sum, sumNumber, updateHeartbeatConnectionFlag)
	local updateHeartbeatConnectionState = nil
	local secondaryHeartbeatConnectionState = nil
	local alternateHeartbeatConnectionState = nil
	local additionalHeartbeatConnectionState = nil
	local fallbackUpdateHeartbeatConnectionState = nil
	local number = updateHeartbeatConnectionFlag or 0.25

	if connection then
		connection:Disconnect()
	end

	local capturedSum = updateInstancePropertiesNumber
	local capturedNumber = updateHeartbeatConnectionNumber
	local timestamp = tick()

	connection = RunService.Heartbeat:Connect(function()
		local differenceNumber = math.min((tick() - timestamp) / number)
		local difference = 1 - (1 - differenceNumber) * (1 - differenceNumber)

		updateInstancePropertiesNumber = capturedSum + (sum - capturedSum) * difference
		updateHeartbeatConnectionNumber = capturedNumber + (sumNumber - capturedNumber) * difference
		updateHeartbeatConnection()

		if difference >= 1 then
			connection:Disconnect()
			connection = nil
			updateInstancePropertiesNumber = sum
			updateHeartbeatConnectionNumber = sumNumber
			updateHeartbeatConnection()
		end
	end, updateHeartbeatConnectionState, secondaryHeartbeatConnectionState, alternateHeartbeatConnectionState, additionalHeartbeatConnectionState, fallbackUpdateHeartbeatConnectionState)
end
